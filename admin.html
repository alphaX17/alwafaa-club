function renderGuidanceList(list) { const container = document.getElementById('guidanceListContainer'); const badge = document.getElementById('guidanceCountBadge'); if (!container) return; // مطابقة المفاتيح الحقيقية الموجودة في Google Sheets const validList = (list || []).filter(item => { const major = (item['الكلية أو الاختصاص'] || item['التخصص'] || '').trim(); const name = (item['الاسم أو الصفة'] || item['مسؤول التخصص'] || '').trim(); const link = (item['رابط الواتساب'] || item['رابط المجموعة'] || '').trim(); return major !== '' || name !== '' || link !== ''; }); liveGuidanceList = validList; if (badge) badge.textContent = validList.length; if (validList.length === 0) { container.innerHTML = `
لا توجد مجموعات توجيه مسجلة حالياً.
`; return; } container.innerHTML = validList.map((item, idx) => { // قراءة القيم حسب ترويسة الجدول في الشيت const major = item['الكلية أو الاختصاص'] || item['التخصص'] || 'عام'; const type = item['النوع'] || 'توجيه طلابي'; const leader = item['الاسم أو الصفة'] || item['مسؤول التخصص'] || 'مسؤول التخصص'; const phone = item['رقم الهاتف'] || item['هاتف المسؤول'] || ''; const link = item['رابط الواتساب'] || item['رابط المجموعة'] || '#'; const directWa = getDirectWhatsAppChatUrl(phone || link); return `
${type}

${major}

المسؤول: ${leader} ${phone ? `• الهاتف: ${phone}` : ''}

رابط الواتساب ${directWa ? ` واتساب المسؤول ` : ''}
`; }).join(''); lucide.createIcons(); } // إرسال الحذف بالبيانات الحقيقية المطابقة للشيت function deleteGuidanceItem(idx) { const target = liveGuidanceList[idx]; if (!target) return; const targetName = target['الاسم أو الصفة'] || target['مسؤول التخصص'] || ''; const targetMajor = target['الكلية أو الاختصاص'] || target['التخصص'] || ''; openGenericDeleteModal(`هل أنت متأكد من حذف سجل (${targetName} - ${targetMajor})؟`, async () => { liveGuidanceList.splice(idx, 1); localStorage.setItem('nadi_guidance_cache', JSON.stringify(liveGuidanceList)); renderGuidanceList(liveGuidanceList); try { await fetch(SCRIPT_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "delete_guidance", major: targetMajor, name: targetName, // إرسال المفاتيح بالصيغتين لضمان مطابقة شرط الحذف في Apps Script "الكلية أو الاختصاص": targetMajor, "الاسم أو الصفة": targetName }) }); } catch(e) {} }); }
